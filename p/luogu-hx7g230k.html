Peaceful Rooks

因为棋子是车，所以移动的先后顺序没有关系。

就从前往后移动就行了。

int n,m;
int fa[N];
inline int find(int x){
    if(x==fa[x])return x;
    return fa[x]=find(fa[x]);
}
signed main(){
    int T=read();
    while(T--){
        n=read();m=read();
        up(i,1,n)fa[i]=i;
        int ans=m,u,v;
        up(i,1,m){
            u=read();v=read();
            if(u==v)ans--;
            else if(find(u)==find(v)){
                ans++;
            }
            else fa[find(u)]=find(v);
        }
        write(ans,1);
    }
    return 0;
}

Cow and Snacks

考虑一个大小\(>1\)的联通块。肯定有一个动物得吃到 \(2\) 种点心，而总有一种排队顺序使得这个块中没有任何一头动物再吃到 \(2\) 种点心，那么一个大小为 \(c\) 的联通块可以满足 \(c−1\) 个动物。

int n,m;
int fa[N];
inline int find(int x){
    if(x==fa[x])return x;
    return fa[x]=find(fa[x]);
}
signed main(){
    n=read();m=read();
    up(i,1,n+m)fa[i]=i;
    int u,v;
    up(i,1,m){
        u=read();v=read();
        fa[find(i)]=find(u+m);
        fa[find(i)]=find(v+m);
    }
    int num=0;
    up(i,1,n+m){
        if(fa[i]==i)num++;
    }
    cout<<m-n+num;
    return 0;
}

Masha-forgetful

struct str{

int T,n,a[1005][1005],m,fr[1005],f[1005],ans[1005],b[1005],rec[1005],cnt;
int hash(int num1,int num2,int num3){
	if(num1==0)num1=10;if(num2==0)num2=10;if(num3==0)num3=10;
	if(num1==-1)num1=0;
	return num1*121+num2*11+num3;
}
void init(){
	cnt=0;
	for(int i=1;i<=3000;i++)t[i].id=t[i].l=t[i].r=0;
	for(int i=1;i<=1000;i++)fr[i]=f[i]=ans[i]=b[i]=rec[i]=0;
}
int main(){
	scanf("%d",&T);
	while(T--){
		init();

    stack<int>s1,s2;

inline bool dfs(int u, int p, int r) {
    if(rt[u]==r)return 0;

inline void dfs(int u){
    up(i,0,6){
        if(a[u][i]>0){
            a[u][i]--;a[i][u]--;
            dfs(i);
            ansx[++cnt]=u;
            ansy[cnt]=i;
        }
    }
}
signed main(){
	cin>>n;
    up(i,1,n){
        cin>>x[i]>>y[i];
        a[x[i]][y[i]]++;a[y[i]][x[i]]++;
        in[x[i]]++;in[y[i]]++;
    }
    int j=0;
    up(i,0,6)if(in[i]>0)j=i;
    up(i,0,6)if((in[i]&1)==1)j=i,cnt++;
    if (cnt==1||cnt>2){
        printf("No solution\n");
        return 0;
    }
    cnt=0;
    dfs(j);
    if(cnt<n){
        printf("No solution\n");
        return 0;
    }
    dn(i,cnt,1){
        up(j,1,n){

Cycle in Graph

int n,m,k;
vector<int>g[N];
int vis[N],pos;
inline void dfs(int u,int tim){
	vis[u]=tim;
	for(auto v:g[u]){
		if(!vis[v]){
			dfs(v,tim+1);
			break;
		}
		if(vis[u]>=vis[v]+k){
			cout<<vis[u]-vis[v]+1<<endl;
			cout<<u<<" ";
			pos=v;
			return;
		}
	}
	if(pos!=-1)cout<<u<<" ";
	if(pos==u)pos=-1;
}

int n,m;
vector<int>g[N];
vector<int>f[N];
bool vis[N];
inline void dfs(int u,int from){
    vis[u]=1;
    for(auto v:g[u]){
        if(vis[v]){
        	if(v!=from)f[v].push_back(u);
            continue;
        }
        dfs(v,u);
        f[u].push_back(v);
        if(f[v].size()){
            printf("%d %d %d\n",f[v][0],v,u);
            f[v].pop_back();
            f[u].pop_back();
        }
    }
    int siz=f[u].size()-1;
    while((siz+1)>1){
        printf("%d %d %d\n",f[u][siz],u,f[u][siz-1]);
        siz-=2;
        f[u].pop_back();
        f[u].pop_back();
    }
}

Wormhouse

用类似数位 DP 的方法来处理。

int n,m;
int g[550][550];
int a[N],ans[N];
inline bool dfs(int u,int w,int lim){
    ans[u]=w;
    if(u==m+1&&w==a[m+1]&&!lim)return 1;
    if(u==m+1){
        ans[u]=0;
        return 0;
    }
    up(i,1,n){
		if((!g[w][i])||(lim&&i<a[u+1])) continue;
		g[w][i]=g[i][w]=0;
		if(dfs(u+1,i,lim&&(i==a[u+1])))return 1;
		g[w][i]=g[i][w]=1;
	}
	ans[u]=0;
    return 0;
}
signed main(){
    cin>>n>>m;
    int u,v;
    up(i,1,m+1){
        cin>>a[i];
        if(i<=1)continue;
        g[a[i]][a[i-1]]++;
        g[a[i-1]][a[i]]++;
    }
    if(dfs(1,a[1],1))up(i,1,m+1)cout<<ans[i]<<" ";
    else puts("No solution");
    return 0;
}
