P1292

简单题，答案为\(a,b\) 的 \(\gcd\)。

然后解方程 \(by\equiv 1\mod (a)\)

\(by-ax=1\)

inline void exgcd(int a,int b,int& d,int& x,int& y){
	int t;
	if(b==0){d=a;x=1;y=0;}
	else{
		exgcd(b,a%b,d,x,y);
		t=x;x=y;y=t-(a/b)*y;
	}
}
int a,b;
signed main(){
    a=read();b=read();
    int gcd=__gcd(a,b);
    cout<<gcd<<endl;
    a/=gcd;b/=gcd;
    int x,y,d;
    exgcd(a,b,d,x,y);
    while(x>0||y<=0){
        x-=b;y+=a;
    }
    while(x+b<=0&&y-a>0)x+=b,y-=a;
    cout<<-x<<" "<<y<<endl;
    return 0;
}

T2

猜出来的结论题。

自己用数据模拟了下，感觉是根据点的权值由大到小一个个删除的。

其实把点删完，等价于把边删完。

删一条边 \([u,v]\) 的权值等于\(\min(a[u],a[v])\)。

所以删最大点一定最优。

所以只用把点的权值从大往小删除就行了。

int n,m;
struct node{
	int v,w,id;
}a[N];
vector<int>g[N];
inline bool cmp(node x,node y){
	return x.v>y.v;
}
unordered_map<int,int>mp;
signed main(){
	n=read();m=read();
	up(i,1,n)a[i].v=read(),a[i].id=i;
	int u,v;
	up(i,1,m){
		u=read();v=read();
		g[u].push_back(v);
		g[v].push_back(u);
	}
	up(i,1,n){
		for(auto v:g[i]){
			a[i].w+=a[v].v;
		}
	}
	sort(a+1,a+1+n,cmp);
	up(i,1,n){
		mp[a[i].id]=i;
	}
	int ans=0;
	up(i,1,n){
		int u=a[i].id;
		ans+=a[i].w;
		for(auto v:g[u]){
			a[mp[v]].w-=a[i].v;
		}
	}
	cout<<ans;
	return 0;
}

inline void bfs(int k,int sx,int sy){
    queue<node>q;
    memset(dis[k],0x3f,sizeof dis[k]);
    dis[k][sx][sy]=0;
    q.push({sx,sy});
    int l=0,r=s,w=0;
    while(l<=r){
        int mid=(l+r)>>1;

            l=0,r=s,w=0;
            while(l<=r){
                int mid=(l+r)>>1;
                if(check(tx,ty,mid)){
                    l=mid+1;w=mid;
                }
                else r=mid-1;
            }
            if(dis[k][tx][ty]==dis[k][u.x][u.y]+1){
                tg[k][tx][ty]=max(tg[k][tx][ty],tg[k][u.x][u.y]+w);

                dis[k][tx][ty]=dis[k][u.x][u.y]+1;
                tg[k][tx][ty]=max(tg[k][tx][ty],tg[k][u.x][u.y]+w);   
                q.push({tx,ty});
            }
        }
    }
}

T4

数据非常小，所以可以直接暴力拆点跑最短路。

\(dis[u][i]\) 表示到了节点 \(u\) ,\(\gcd\) 为 \(i\) 的最短路。

inline void add(int u,int v,int w){
	up(i,0,500){
		int x=__gcd(w,i);
		g[u][x].push_back({v,i,w/x});
		g[v][x].push_back({u,i,w/x});

inline void spfa(){
	queue<pii>q;
	memset(dis,0x3f,sizeof dis);
	up(i,0,500){
		q.push({n,i});
		dis[n][i]=0;
		vis[n][i]=1;

	while(q.size()){
