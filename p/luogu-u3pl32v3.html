P2762 太空飞行计划问题

题目不说人话

大意：

有\(m\)个实验，每个实验只可以进行一次，但会获得相应的奖金，有\(n\)个仪器，每个实验都需要一定的仪器，每个仪器可以运用于多个实验，但需要一定的价值，问奖金与代价的差的最大值是多少？

模型：

给定一张图，有左侧的点和右侧的点，左侧的点点权为正（对应试验），右侧的点点权为负（对应器材），如果选择了左侧的某个点就必须要选右边的一部分点。要求最大化点权和。

如果将左侧的点和右侧的点之间对应连边，如果该实验要求该器材就连一条边，那么问题就被转化为了这样一个问题：

给定一个有向图，点有点权，选择一个子图，满足子图上如果选择了一个点就必须选择它后继的所有点。最大化点权和。

这是一个经典的网络流问题，如果一个点被选择了则后继必须被选择，那么称该图是 闭合的，因此该问题叫做最大权闭合子图问题。可以使用最小割解决。

P5038 [SCOI2012] 奇怪的游戏

黑白染色，二分答案。

inline bool check(int val) {
	E.clear();
	int sum = 0;
	up(i,1,n) {
		up(j,1,m){
			if((i+j)%2==0) {
				E.add(s,id[i][j],val-a[i][j]);
				sum+=val-a[i][j];
				up(k,0,3){
					int tx=i+dx[k];
					int ty=j+dy[k];
					if(tx >= 1 && tx <= n && ty >= 1 && ty <= m) {
						E.add(id[i][j],id[tx][ty],inf);
					}
				}
			}
			else {
				E.add(id[i][j],t,val-a[i][j]);
			}
		}
	}
	return E.maxflow(s,t)==sum;
}
signed main(){
	int T=read();
	while(T--){
		int cnt=0;
		memset(a,0,sizeof a);
		memset(id,0,sizeof id);
		n=read();m=read();
		int nb=0,nw=0,tb=0,tw=0,maxl=0;
		t=n*m+10;s=0;
		up(i,1,n){
			up(j,1,m){
				a[i][j]=read();
				maxl=max(maxl,a[i][j]);
				id[i][j]=++cnt;
				if((i+j)%2==0){
					nb+=a[i][j];
					tb++;
				}
				else{
					nw+=a[i][j];
					tw++;
				}
			}
		}
		if(tw!=tb){
			int x=(nb-nw)/(tb-tw);
			if(x>=maxl&&check(x)) {
				cout<<x*tb-nb<<endl;
			}
			else {
				cout<<-1<<endl;
			}
			continue;
		}
		if(nw!=nb) {
			cout<<-1<<endl;
			continue;
		}
		int l=maxl,r=inf,ans;
		while(l<=r){
			int mid=(l+r)>>1;
			if(check(mid)){
				ans=mid;
				r=mid-1;
			}
			else l=mid+1;
		}
		cout<<ans*tb-nb<<"\n";
	}
	return 0;
}
