结论题：最优情况一定只有两个点

证明：假设有一点对，从这点对延伸出去的一条边可以使点权与边权的比变大；

设原先点的边权为\(a\)，点权为\(ka\)，比值为\(k\)，则延伸后比值为\(k+d\)，边值为\(a+l\)，则点权为（\(a+l)\)*（\(k+d\)）=\(ak+ad+lk+ld\)，增加点权为 \(ad+lk+ld\) 则延伸边的比为(\(k+d+ad/l\)）> \(k+d\)

\(Qed;\)

题意比较简单，想出暴力应该不难画出函数图像，当距离最小时，一定是速度最快的豹与一个豹的交点；

#include<bits/stdc++.h>

#define int long long

	return l.v<r.v;
}

	return (double)(a[i].v*a[i].t-a[j].v*a[j].t)/(a[i].v-a[j].v);
}
signed main(){
	//freopen("chase.in","r",stdin);
	//freopen("chase.out","w",stdout);
	n=read();
	for(int i=1;i<=n;i++){
		a[i].t=read();
		a[i].v=read();
		maxt=max(a[i].t,maxt);
	}
	sort(a+1,a+1+n,cmp);
	double ans=0,maxl,minl=1e14;
	for(int i=1;i<=n;i++){
		maxl=max(maxl,a[i].v*(maxt-a[i].t));
		minl=min(minl,a[i].v*(maxt-a[i].t));
	}
	ans=maxl-minl;
	for(int l=1;l<n;l++){
		double jx=clac(n,l);
		if(jx<maxt)continue;
		maxl=0;minl=1e14;
		for(int i=1;i<=n;i++){
			maxl=max(maxl,a[i].v*jx-a[i].v*a[i].t);
			minl=min(minl,a[i].v*jx-a[i].v*a[i].t);
		}
		if(maxl==a[n].v*jx-a[n].v*a[n].t)break;

在进一步分析，最大距离其实是一个单峰函数，在某一点后只会上升

所以只用三分法求定点

#include<bits/stdc++.h>

using namespace std;

#define int long long

const double eps=1e-8;

inline int read(){
	int x=0,f=1;char ch=getchar();
	while (ch<'0'||ch>'9'){if(ch=='-') f=-1;ch=getchar();}
	while (ch>='0'&&ch<='9'){x=x*10+ch-48;ch=getchar();}
	return x*f;
}
struct node{
	int t,v;
}a[500500];
int n;
double l=0;
double check(double mid){
	double minn,maxx,temp;
	minn=maxx=(mid-a[1].t)*a[1].v;
	for(int i=2;i<=n;i++){
		temp=(mid-a[i].t)*a[i].v;	
		minn=min(minn,temp);
		maxx=max(maxx,temp);
	}
	return maxx-minn;
}
signed main(){
	n=read();
	for(int i=1;i<=n;i++){
		a[i].t=read();
		a[i].v=read();
		l=max(l,(double)a[i].t);
	}
	double r=1e15,lmid,rmid;
	while(r-l>eps){
		lmid=l+(r-l)/3;
		rmid=r-(r-l)/3;
		if(check(lmid)<check(rmid))r=rmid;
		else l=lmid;
	}
	printf("%.2lf",check(l));
    return 0;
}
