\(T1\)

差分约束，但是不能一遍\(spfa\)直接 $ dis[i]-dis[j] $,

看n的范围\(n\le200\),考虑弗洛伊德更新二维\(dis\)数组。

int main(){
	ios_base::sync_with_stdio(false), cin.tie(0), cout.tie(0);
	cin >>n >>m;
	for(int i = 1; i <= n; i ++) {
		for(int j = 1; j <= n; j ++) {
			g[i][j] = Inf;
		}
	}
	for(int i = 1, u, v, w, op; i <= m; i ++) {
		cin >>op >>u >>v >>w;
		if(op == 0) {
			g[v][u] = min(g[v][u], (i64)w);
		}
		else if(op == 1) {
			g[u][v] = min(g[u][v], (i64)-w);
		}
		else {
			g[v][u] = min(g[v][u], (i64)w);
			g[u][v] = min(g[u][v], (i64)-w);
		}
	}
	for(int k = 1; k <= n; k ++) {
		for(int i = 1; i <= n; i ++) {
			for(int j = 1; j <= n; j ++) {
				if(g[i][k] >= Inf || g[k][j] >= Inf) continue;
				g[i][j] = min(g[i][j], g[i][k] + g[k][j]);
			}
		}
	}
	cin >>q;
	for(int i = 1, x, y; i <= q; i ++) {
		cin >>x >>y;
		if(g[y][x] >= Inf) {
			cout <<"Infinity" <<endl;
		}
		else {
			cout <<g[y][x] <<endl;
		}
	}
}

\(T2\)

没什么好讲的，\(dis[u][0]\) 为进 \(u \) 的距离，\(dis[u][1]\) 为出\(u\)的距离，

\(spfa\) 被卡了，所以用 \(djs\) 。

int n,m,s,Q;
int re[250500],gr[250500];
vector<pii>g[250500];
int dis[250500][2];
bool vis[250500];
priority_queue<pii,vector<pii>,greater<pii> >q;
inline void djs(){
	memset(dis,0x3f,sizeof dis);
	dis[s][1]=re[s];
	dis[s][0]=re[s];
	q.push({dis[s][1],s});
	while(q.size()){
		int u=q.top().second;q.pop();
		if(vis[u])continue;
		vis[u]=1;
		for(auto i:g[u]){
			int v=i.first,w=i.second;
			if(dis[v][0]>dis[u][1]+w){
				dis[v][0]=dis[u][1]+w;
				if(dis[v][0]%(re[v]+gr[v])>re[v])dis[v][1]=dis[v][0];
				else dis[v][1]=dis[v][0]+re[v]-dis[v][0]%(re[v]+gr[v]);
				q.push({dis[v][1],v});
			}
		}
	}
}
signed main(){
	n=read();m=read();s=read();Q=read();
	for(int i=1;i<=n;i++){
		re[i]=read();
		gr[i]=read();
	}
	while(m--){
		int u=read(),v=read(),w=read();
		g[u].push_back({v,w});
		g[v].push_back({u,w});
	}
	djs();
	cout<<dis[Q][0];
	return 0;
}

inline void spfa1(){
	q.push(0);

	dis[0][1]=dis[0][0]=0;
	while(q.size()){
		int u=q.front();q.pop();

inline void dfs(int u){
	if(!u)return;
	for(auto v:e[u]){
		if(dis[v][1]==dis[u][0]){
			edge[u].push_back(v);
			edge[v].push_back(u);
			dfs(v);
		}
	}
}
int dfn[155000],idx,low[155000];
bool num[155000];
vector<int>ans;
inline void dfs_cut(int u,int fa){
	dfn[u] = low[u] = ++idx;
	int child = 0;
	for(auto v:edge[u]) {
		if(!dfn[v]){
			dfs_cut(v, fa);
			low[u] = min(low[v], low[u]);
			if(low[v] >= dfn[u] && u != fa) ans.push_back(u);
			if(u == fa) child ++;
		}
		low[u] = min(low[u], dfn[v]);
	}
	if(child >= 2 && u == fa) ans.push_back(u);
}
