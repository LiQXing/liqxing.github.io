\(T1\)

sbsbsbsbsbsbsbsbsbsbsb。

最难的一道题。

因为边的权值小于 \(30\)。 所以可以暴力枚举最大的边权。

然后让这条边的次数尽可能的多。

把联通块分成多个，使得匹配最多。

另 \(x=\max \{a_i\},s=\sum a_i\)

如果 \(x\times 2>s\)，有 \(s-x\) 个匹配。

否则,有 \(s/2\) 个匹配。

那么就一次一次删点就完了。

int n,m,k;
int ans;
inline int ksm(int a,int b){
    int res=1;
    while(b){
        if(b&1)res=res*a%mod;
        a=a*a%mod;
        b>>=1;
    }
    return res;
}
vector<pii>g[N];
vector<int>p[N];
bool vis[N];
int del[N],cnt,lim;
inline void dfs(int u){
    vis[u]=1;
    p[cnt].push_back(u);
    for(auto [v,w]:g[u]){
        if(vis[v]||del[v]||w>lim)continue;
        dfs(v);
    }
}
int ex;
signed main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0);cout.tie(0);
    cin>>n;
    int u,v,w;
    up(i,1,n-1){
        cin>>u>>v>>w;
        g[u].push_back({v,w});
        g[v].push_back({u,w});
    }    
    dn(k,30,0){
        memset(vis,0,sizeof vis);
        cnt=0;lim=k-1;
        up(i,1,n){
            if(!del[i]&&!vis[i]){
                cnt++;
                p[cnt].clear();
                dfs(i);
            }
        }
        int maxl=0,pos=0,tot=0;
		up(i,1,cnt){
			int u=p[i].size();
			if(u>maxl){
				maxl=u;
                pos=i;
			}
			tot+=u;
		}
        if((maxl-ex)*2<=tot-ex){
			ans=(ans+ksm(n,k)*(tot-ex-1))%mod;
			break;
		}
		tot-=ex;
        maxl-=ex;
		ans=(ans+2*ksm(n,k)*(tot-maxl))%mod;
		up(i,1,cnt){
			if(i==pos) continue;
			for(auto u:p[i]){
				del[u]=1;
			}
		}
		ex+=tot-maxl;
    }
    cout<<ans;
    return 0;
}

\(T3\)

有意思的一道题。

首先如果是| ^ 那么一定是把所有祖先都选上。

|是显然的，对于^，有 \(a\oplus c=a\oplus b\oplus b\oplus c\le a\oplus b+b\oplus c\)

至于&

可以枚举最高位 \(1\) 为 \(k\) 的数，只用考虑最近的点就行了，有点像树上背包。

\(T4\)

\(f[i][j]\) 表示 \(i\) 个数，和为 \(j\)。 显然：

\(f[i][j]=f[i-1][j-i]+f[i-1][j-1]\)

设 \(a_i\) 出现了 \(x\) 次。

贡献为 \(a_i^m\times f[i-x][n-x\times a_i]\)

int n,m,k;
int ans;
inline int ksm(int a,int b){
    int res=1;
    while(b){
        if(b&1)res=res*a%mod;
        a=a*a%mod;
        b>>=1;
    }
    return res;
}
int f[5050][5050];
signed main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0);cout.tie(0);
    cin>>n>>k>>m;
    f[0][0]=1;
    up(i,1,k){
        up(j,i,n){
            f[i][j]=(f[i][j-i]+f[i-1][j-1])%mod;
        }
    }
    int ans=0;
	up(i,1,n) {
		int s=0;
		for (int j=1;j<=k&&j*i<=n;j++)s=(s+f[k-j][n-i*j])%mod;
		ans=(ans+s*ksm(i,m)%mod)%mod;
	}
	cout<<ans;
    return 0;
}
