The Awesomest Vertex

树上问题，习惯性的拍成 dfs 序。 因为有绝对值，所以可以化成 \(\max(a[i],-a[i])\)。 \(b\) 是个定值，所以可以预处理出来。 每次修改可以类似吉老师线段树那样去维护。

int n,q;
int a[N],b[N],p[N];
vector<int>g[N];
int cnt;
int in[N*2],out[N*2];
struct node{
	int k,b;
}val[N*2];
struct node2{
    int l,r,k,b;
}tr[N<<2];
int lim[N<<2],d[N<<2];
inline void dfs(int u,int from){
    in[u]=++cnt;
    a[u]+=a[p[u]];
    b[u]+=b[p[u]];
    int v=abs(b[u]);
	val[cnt]={v,a[u]*v};
    for(auto v:g[u]){
        if(v==from)continue;
        dfs(v,u);
    }
    out[u]=++cnt;
    val[cnt]={-v,-a[u]*v};
}
inline void push_up(int k){
    int v=(tr[lc].b>=tr[rc].b?(lc):(rc));
	tr[k].k=tr[v].k;
	tr[k].b=tr[v].b;
	lim[k]=min((tr[v].k>=tr[v^1].k?inf:(tr[v].b-tr[v^1].b)/(tr[v^1].k-tr[v].k)),min(lim[lc],lim[rc]));
}
inline void build(int k=1,int l=1,int r=cnt){
    tr[k].l=l;tr[k].r=r;
    if(l==r){
        tr[k].k=val[l].k;
        tr[k].b=val[l].b; 
		lim[k]=inf;
		return;
	}
	int mid=(l+r)>>1;
	build(lc,l,mid);
	build(rc,mid+1,r);
	push_up(k);
}
inline void upd(int k,int v){
    tr[k].b+=tr[k].k*v;
    lim[k]-=v;
    d[k]+=v;
}
inline void push_down(int k){
    if(!d[k])return;
    upd(lc,d[k]);
    upd(rc,d[k]);
    d[k]=0;
}
inline int ask(int k,int l,int r){
    if(tr[k].l>=l&&tr[k].r<=r)return tr[k].b;
    push_down(k);
    int mid=(tr[k].l+tr[k].r)>>1;
    int ret=-inf;
    if(l<=mid)ret=ask(lc,l,r);
    if(r>mid)ret=max(ret,ask(rc,l,r));
    return ret;
}
inline void change(int k,int l,int r,long long v){
    if(!tr[k].r)return;
    if(tr[k].l>=l&&tr[k].r<=r&&v<=lim[k]){
        upd(k,v);
        return; 
    }
    push_down(k);
    int mid=(tr[k].l+tr[k].r)>>1;
    if(l<=mid)change(lc,l,r,v);
    if(r>mid)change(rc,l,r,v);
    push_up(k);
}
signed main() {
    n=read();q=read();
    up(i,2,n){
        p[i]=read();
        g[p[i]].push_back(i);
        //g[i].push_back(p[i]);
    }
    up(i,1,n)a[i]=read();
    up(i,1,n)b[i]=read();
	dfs(1,0);
    build();
    int op,v,x;
    while(q--){
        op=read();v=read();
		if(op==1){
			x=read();
			change(1,in[v],out[v],x);
		}
		else{
			printf("%lld\n",ask(1,in[v],out[v]));
		}
    }
    return 0;
}

Dogeforces

祖先的权值严格大于当前点。

根节点就很好判断，以下每一个子树同理。

所以可以递归处理每一个子树。

int n,m;
int l[550][550];
int a[N],vst[N],cnt;
inline int findrt(vector<int>s) {
	int mx=0;
	for(auto u:s) for(auto v:s) if(l[u][v]>mx) mx=l[u][v];
	return mx;
}
int fa[N];
inline int solve(vector<int>s) {
	if(s.size()==1) return s[0];
	int sz=s.size(),u=findrt(s),id=++cnt;
	a[id]=u;
	for(auto v:s) vst[v]=0;
	for(auto v:s) if(!vst[v]) {
		vector<int>t;
		for(auto w:s) if(l[v][w]!=u) vst[w]=1,t.push_back(w);
		fa[solve(t)]=id;
	}
	return id;
}
signed main(){
    n=read();cnt=n;
    up(i,1,n){
        up(j,1,n){
            l[i][j]=read();
        }
    }
    up(i,1,n)a[i]=l[i][i];
    vector<int>s;
    up(i,1,n)s.push_back(i);
    int rt=solve(s);
    write(cnt,1);
	up(i,1,cnt)write(a[i],0);
    puts("");
	write(rt,1);
	up(i,1,cnt)if(i!=rt)printf("%lld %lld\n",i,fa[i]);
    return 0;
}

Minimal Labels

拓扑好题。

因为要求字典序最小，所以直接用优先队列维护。

int n,k,m;
int a[N];
vector<int>g[N];
int id[N],tms,deg[N];
inline void topo() {
	priority_queue<int> q;
	up(i,1,n) if(!deg[i]) q.push(i);
	while(!q.empty()) {
		int u = q.top(); q.pop();
		id[u] = ++tms;
		for(auto v:g[u]) if(!--deg[v]) q.push(v);
	}
}

signed main() {
	
    n=read();m=read();
	int u,v;
    up(i,1,m){
		u=read();v=read();
		g[v].push_back(u);
		++deg[u];
	}
	topo();
	up(i, 1, n) printf("%d%c", n+1-id[i]);
	return 0;
}

Three States

暴力bfs判断。

int n,m;
char a[1500][1500];
int dis[1500][1500][4];
int dx[]={0,0,1,-1},dy[]={1,-1,0,0};
inline void bfs(int s){
    deque<pii>q;
    up(i,1,n){
        up(j,1,m){
            if(a[i][j]==s+'0'){
                dis[i][j][s]=0;
                q.push_back({i,j});
            }
        }
    }
    while(q.size()){
        pii u=q.front();q.pop_front();
        up(i,0,3){
            int tx=u.first+dx[i],ty=u.second+dy[i];
            if(tx<1||tx>n||dis[u.first][u.second][s]+1>=dis[tx][ty][s]||ty<1||ty>m||a[tx][ty]=='#')continue;
			if(a[tx][ty]=='.'){
				dis[tx][ty][s]=dis[u.first][u.second][s]+1;
				q.push_back({tx,ty});
			}
			else{
				dis[tx][ty][s]=dis[u.first][u.second][s];
				q.push_front({tx,ty});
		    }
        }
    }
}
signed main() {
    n=read();m=read();
    up(i,1,n){
        scanf("%s",a[i]+1);
    }
    memset(dis,0x1f,sizeof dis);
    bfs(1);bfs(2);bfs(3);
    int minl=inf;
    up(i,1,n){
		up(j,1,m){
			int cost=dis[i][j][1]+dis[i][j][2]+dis[i][j][3];
			if(a[i][j]=='.') cost-=2; 
			minl=min(minl,cost);
		}
	}
	if(minl>=inf)cout<<"-1";
	else cout<<minl;
    return 0;
}
