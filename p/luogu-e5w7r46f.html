    int cnt=0,root=0;
    struct node{

        int key,pri;
        int siz,num;

    }tr[N];
    int ans;
    inline void build(int x,int num){
        cnt++;
        tr[cnt].siz=num;
        tr[cnt].num=num;
        tr[cnt].ls=tr[cnt].rs=0;
        tr[cnt].sum=x*num;
        tr[cnt].sum2=x*num;
        tr[cnt].key=x;
        tr[cnt].pri=rand();
    }
    inline void push_up(int k){
        tr[k].siz=tr[tr[k].ls].siz+tr[tr[k].rs].siz+tr[k].num;
        tr[k].sum=tr[tr[k].ls].sum+tr[tr[k].rs].sum+tr[k].sum2;
    }
    inline void split(int u,int x,int &l,int &r){
        if(u==0){l=r=0;return;}
        if(tr[u].key<=x){
            l=u;
            split(tr[u].rs,x,tr[u].rs,r);
        }
        else{
            r=u;
            split(tr[u].ls,x,l,tr[u].ls);
        }
        push_up(u);
    }
    inline int merge(int l,int r){
        if(!l||!r)return l|r;
        if(tr[l].pri>tr[r].pri){
            tr[l].rs=merge(tr[l].rs,r);
            push_up(l);
            return l;
        }
        else{
            tr[r].ls=merge(l,tr[r].ls);
            push_up(r);
            return r;
        }
    }

    	int l,r,p;

        split(l,x-1,l,p);
        if(tr[p].num==0){
			build(x,nums);
			p=cnt; 
		} 
		else{
			tr[p].num+=nums;
			tr[p].siz+=nums;
			tr[p].sum+=nums*tr[p].key;
			tr[p].sum2+=nums*tr[p].key;
		}
        root=merge(merge(l,p),r);
	}
    inline void del(int x,int nums){
        int l,r,p;
        split(root,x,l,r);
        split(l,x-1,l,p);
        tr[p].siz-=nums;
        tr[p].num-=nums;
        tr[p].sum-=tr[p].key*nums;
        tr[p].sum2-=tr[p].key*nums;

\(T3\)

暴力还是比较好想的,直接每加一个区间都暴力建边，然后用冰茶几维护连通性。

复杂度\(O(n^2)\)

如何优化？

首先离散化，把每个区间缩小；

考虑连边的两种情况：第一种是包含，小的向大的连边；第二种是相交不包含，连双向边。

可以使用线段树来优化。把区间拆放到 \(\log\) 个线段树结点上。

加新区间连边就把左右端点对应线段树叶子结点到根到路径上所有放的路径都合并掉。

\(T4\)

强制在线求n个点\((x,y)\)的切比雪夫距离和。（带修改）

引理：

曼哈顿距离与切比雪夫距离的转换

证明1：

发现离原点 \((0,0)\) 曼哈顿距离为 \(x\) 的点构成的集合刚好是一个偏了\(45\) 的正方形

把这个正方形转正，再扩大 \(\sqrt 2\) 倍，得到的新的正方形就是距原点切比雪夫距离为 \(x\)的点的集合

于是就有 \((x,y) \) 变到\((x+y,x-y)\)

切比雪夫转为曼哈顿：\((x,y) \) 变到\(((x+y)/2,(x-y)/2)\)

证明2：

曼哈顿：\(|x-x_i|+|y-y_i|\)

拆开看，有四种

  1. \(x-x_i+y-y_i\).

  2. \(x-x_i-y+y_i\)

  3. \(-x+x_i-y+y_i\)

  4. \(-x+x_i+y-y_i\)

取最大值

而这恰恰就是 \((x+y,x-y)\)到 \((x_i+y_i,x_i-y_i)\)的切比雪夫距离

把切比雪夫距离转化为曼哈顿距离后就好做多了。

用两颗平衡树储存x方向和y方向

		t+=tr[r].sum-x*tr[r].size;
		root=merge(l,r);
		return t;
	}

一个非常显然的事实：一个点双对答案的贡献必然是点双里面的最小权值

建立圆方树，方点的权值为点双中的最小圆点权值。

然后原图就变成了一棵树，询问时就可以直接树剖套线段树求路径最小值了.

对于一个方点，\(multiset\)里面存它所有子节点的权值。

然后修改一个圆点时，就只需要动它父亲的multiset（它的父亲必然是一个方点）。
