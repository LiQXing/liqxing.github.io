int n;
inline int get_phi(int x){
	int res=x,tmp=x;
	if(x==1)return 1;
	for(int i = 2; i * i<=x; i ++) {
		if(tmp % i == 0) {
			res = res / i * (i - 1);
			while(tmp % i == 0) tmp /= i;
		}
	}
	if(tmp > 1) res = res / tmp * (tmp - 1);
	return res;
}

T2

枚举质因子个数。

简单 \(dfs\)，不要看错数据。

int n;
int val[]={0,2,3,5,7,11,13,17,19,23};
inline int ksm(int a,int b){
    int res=1;
    while(b){
        if(b&1)res=res*a;
        a=a*a;
        b>>=1;
    }
    return res;
}
int ans;
int maxl;
inline void dfs(int u,int now,int lim,int tt){
    if(now>n)return;
    if(u==10){
        if(ans<=tt){
            if(ans==tt)maxl=min(maxl,now);
            else {
                ans=tt;
                maxl=now;
            }
        }
        return;
    }
    up(i,0,lim){
        dfs(u+1,now*ksm(val[u],i),min(i,lim),tt*(i+1));
    }
}
signed main(){
    n=read();
    dfs(1,1,6,1);
    cout<<maxl;
    return 0;
}
