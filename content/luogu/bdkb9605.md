# T1
单调队列优化DP。

显然有一个递推柿子：

$$ dp[i]=\max(dp[j-1]+sum[i]-sum[j])$$

$$dp[i]=sum[i]+\max(dp[j-1]-sum[j])$$

后面的柿子可以用单调队列求出最小值。
```cpp
#include<bits/stdc++.h>

using namespace std;

#define int long long
//#define pii pair<int,int>

const int N=3e4+5;

inline int read(){
	int x=0,f=1;char ch=getchar();
	while(ch<'0'||ch>'9'){if(ch=='-')f=-1;ch=getchar();}
	while(ch>='0'&&ch<='9'){x=x*10+ch-48;ch=getchar();}
	return x*f;
}
int n,k,dp[500500][2];
int sum[500500];
int a[500500];
deque<int>q;
signed main(){
	n=read();k=read();
	for(int i=1;i<=n;i++){
		a[i]=read();
		sum[i]=sum[i-1]+a[i];
	}
	q.push_back(0);
    for (int i = 1; i <=n; i++){
        dp[i][0] = max(dp[i - 1][0], dp[i - 1][1]);
        while (!q.empty() && q.front() < i -k)q.pop_front();
    	dp[i][1] = dp[q.front()][0] + sum[i] - sum[q.front()];
        while(!q.empty() && dp[q.back()][0] - sum[q.back()] < dp[i][0] - sum[i])q.pop_back();
    	q.push_back(i);
    }
    cout << max(dp[n][0], dp[n][1]) << endl;
	return 0;
}
```

```cpp
	int res=1;

		a=(a*a)%p;

inline int C(int n,int m,int p){
	if(m<0)return 0;
```
