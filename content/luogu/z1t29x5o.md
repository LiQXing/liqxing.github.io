```cpp
char s1[N],s2[N];
int nxt[N];
inline void getnxt(){
    int len=strlen(s2+1),j=0;
    up(i,2,len){
        while(j&&s2[i]!=s2[j+1])j=nxt[j];
        if(s2[j+1]==s2[i])j++;
        nxt[i]=j;
    }
}
inline void kmp(){
    int len=strlen(s1+1),j=0,len2=strlen(s2+1);
    up(i,1,len){
        while(j&&s1[i]!=s2[j+1])j=nxt[j];
        if(s2[j+1]==s1[i])j++;
        if(j==len2){
            write(i-j+1,1);
            j=nxt[j];
        }
    }
}
signed main(){
    scanf("%s",s1+1);
    scanf("%s",s2+1);
    getnxt();
    kmp();
    up(i,1,strlen(s2+1)){
        write(nxt[i],0);
    }
    return 0;
}
```

### [String Factoring](https://www.luogu.com.cn/problem/UVA11022)

非常妙的区间 DP。

可以 KMP 优化，但是我写的是暴力。

 首先有第一个决策：
$f[l][r]=\min(f[l][k]+f[k+1][r])$。

第二个决策是 $f[l][r]$  可以压缩的长度。

令压缩后的长度为 $f[l][l+x-1]$ ，如果有循环节，那么可以被压缩。

```cpp
string s,ss[500];
int nxt[N];
int f[500][500];
inline int check(int l,int r){
    up(k,1,r-l+1){
        if((r-l+1)%k)continue ;
        int cnt=0;
        for(int x=l;x<=r;x+=k){
            ss[++cnt] = s.substr(x, k);
            if (cnt>=2&&ss[cnt]!=ss[cnt-1])goto F;
        }
        return k;
        F:;
    }
    return 0;
}
signed main(){
    ios_base::sync_with_stdio(0);
    cin.tie(0);cout.tie(0);
    while(cin>>s&&s!="*"){
        int n=s.size();
        s=" "+s;
        memset(f,0x3f,sizeof f);
        up(i,1,n)f[i][i]=1;
        up(len,2,n){
            up(l,1,n-len+1){
                int r=l+len-1,x=check(l,r);
                f[l][r]=f[l][l+x-1];
                up(k,l,r-1){
                    f[l][r]=min(f[l][r],f[l][k]+f[k+1][r]);
                }
            }
        }
        cout<<f[1][n]<<"\n";
    }
    return 0;
}
```

### [Compress Words](https://www.luogu.com.cn/problem/CF1200E)

```cpp
int n,cnt,top;

int nxt[N];

	n=read();
	up(i,1,n){
		scanf("%s",s+1);
		if(i==1){
			int len=strlen(s+1);
			up(j,1,len)a[++top]=s[j];
		}
		else{
            int lena = top, lenb = strlen(s + 1);
            int nlen = min(lena, lenb), cnt = lenb;
            s[++cnt]='#';
            for(int j = 1; j <= nlen; ++ j)s[++cnt]=a[lena - nlen + j];
            nxt[0] = nxt[1]=0;int j=0;
            for(int k = 2; k <= cnt; ++ k){
                while(j && s[k] != s[j + 1]) j = nxt[j];
                if(s[k] == s[j + 1]) j += 1;
                nxt[k] = j;
            }
            for(int j = nxt[cnt] + 1; j <= lenb; ++ j)a[++top]=s[j];
		}
	}
	up(i,1,top)cout<<a[i];
	return 0;
}
```

```cpp
        Next[i] = j;

        if(chk(i)){
```
