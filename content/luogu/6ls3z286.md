### [Quantity of Strings ](https://www.luogu.com.cn/problem/CF150B)

结论题。

如果 $k$ 为奇数，输出 $m^2$，否则输出 $m$。
然后再加上一些特判。

```cpp
signed main(){
    ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);
    cin>>n>>m>>k;
    if(k>n){
        cout<<ksm(m,n);
        return 0;
    }
    if(k==n){
        cout<<ksm(m,(n+1)/2);
        return 0;
    }
    if(k==1){
        cout<<ksm(m,n);
        return 0;
    }
    if(k%2==1){
        cout<<m*m<<endl;
        return 0;
    }
    else {
        cout<<m;
    }
	return 0;
}
``` 

### [pSort](https://www.luogu.com.cn/problem/CF28B)

并查集神秘题。

```cpp

int n;
int a[N],d[N];
int fa[N];
inline int find(int x){
    if(x==fa[x])return x;
    return fa[x]=find(fa[x]);
}
signed main(){
    ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);
	cin>>n;
    up(i,1,n)cin>>a[i],fa[i]=i;
    up(i,1,n){
        cin>>d[i];
        if(i>d[i])fa[find(i)]=find(i-d[i]);
        if(i+d[i]<=n)fa[find(i)]=find(i+d[i]);
    } 
    up(i,1,n){
        if(fa[find(i)]!=fa[find(a[i])]){
            cout<<"NO";
            return 0;
        }
    }
    cout<<"YES";
    return 0;
}
```

### [Maze](https://www.luogu.com.cn/problem/CF377A)

有一种正难则反的美。

```cpp

int n,m;
char s[555][555];
int sx,sy,num,k,dx[]={0,0,1,-1},dy[]={1,-1,0,0};
bool vis[555][555];
signed main(){
    ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);
	cin>>n>>m>>k;
    up(i,1,n){
        cin>>(s[i]+1);
        up(j,1,m){
            if(s[i][j]=='.')s[i][j]='X',sx=i,sy=j,num++;
        }
    }
    queue<pii>q;
    q.push({sx,sy});
    s[sx][sy]='.';
    vis[sx][sy]=1;
    int cnt=1;
    if(cnt==num-k)goto F;
    while(q.size()){
        pii u=q.front();q.pop();
        int ux=u.first,uy=u.second;
        up(i,0,3){
            int tx=ux+dx[i],ty=uy+dy[i];
            if(tx<1||ty<1||tx>n||ty>m)continue;
            if(vis[tx][ty])continue;
            if(s[tx][ty]=='#')continue;
            vis[tx][ty]=1;
            q.push({tx,ty});
            s[tx][ty]='.';
            cnt++;
            if(cnt==num-k)goto F;
        }
    }
    F:;
    up(i,1,n){
        cout<<(s[i]+1)<<endl;
    }
    return 0;
}
```

### [Suspects](https://www.luogu.com.cn/problem/CF156B)

因为罪犯只有一个，所以可以枚举罪犯，来判断是否合理。

```cpp
int n,m;
int a[N],fake[N],num,vis[N],cnt;
signed main(){
    ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);
	cin>>n>>m;
    up(i,1,n){
        cin>>a[i];
        if(a[i]>0)fake[a[i]]++;
        else fake[-a[i]]--,num++;
    }
    up(i,1,n){
        if(num+fake[i]==m)vis[i]=1,cnt++;
    }
    up(i,1,n){
        if(a[i]>0){
            if(!vis[a[i]]) puts("Lie");
			else if(cnt==1) puts("Truth");
			else puts("Not defined");
        }
        else{
            if(!vis[-a[i]]) puts("Truth");
			else if(cnt==1) puts("Lie");
			else puts("Not defined");
        }
    }
    return 0;
}
```

### [Fedor and Essay](https://www.luogu.com.cn/problem/CF467D)

```cpp
#include<bits/stdc++.h>

class node{
    public:
        int r,len,name;
    bool operator < (const node & n) const {
        if(r==n.r)return len>n.len;
        else return r>n.r;
    } 
};
struct edge {
	int r,len;
}c[100005];
priority_queue<node>q;
vector<int>e[100005]; 
bool vis[100005];
void dijkstra(){
    memset(vis,0,sizeof(vis));
    for(int i=1;i<=cnt;i++) {
    	node t;
    	t.r=c[i].r;
    	t.len=c[i].len;
    	t.name=i;
    	q.push(t);
    }
    while(!q.empty()) {
        int now=q.top().name; 
        q.pop();
        if(vis[now])continue;

        for(int i=0;i<e[now].size();i++) {
            int v=e[now][i];
            if(c[v].r>c[now].r||(c[v].r==c[now].r&&c[v].len>c[now].len))c[v]=c[now];
            node t;
            t.r=c[v].r;
            t.len=c[v].len;
            t.name=v;
            q.push(t);
        }
    }
}
```

### [The Tag Game](https://www.luogu.com.cn/problem/CF813C)

直接暴力跑两遍 dfs，求出 1 号点到每个点的距离和 x 号点到每个点的距离。之后暴力 i 号点和 x 号点到某个点的距离。因为 alice 跑的一定需要比 bob 远（如果 bob 跑得远说明在这之前已经被抓到了），所以答案为 alice 跑的距离*2（在alice没抓到 bob 前 bob 可以随便跑。

```cpp

int n,x;
vector<int>g[N];
int dep[2][N];
inline void dfs(int u,int from,int d[]){
    d[u]=d[from]+1;
    for(auto v:g[u]){
        if(v==from)continue;
        dfs(v,u,d);
    }
}
signed main(){
    ios_base::sync_with_stdio(0);cin.tie(0);cout.tie(0);
    cin>>n>>x;
    int u,v;
    up(i,1,n-1){
        cin>>u>>v;
        g[u].push_back(v);
        g[v].push_back(u);
    }
    dfs(1,0,dep[0]);
    dfs(x,0,dep[1]);
    int ans=0;
    up(i,2,n){
        if(dep[0][i]>dep[1][i]){
			ans=max((dep[0][i]-1)*2,ans);
		}
    }
    cout<<ans;
	return 0;
}
```
