


### [Shortest Path Problem?](https://www.luogu.com.cn/problem/CF845G)

我们来考虑一下没有环的情况，就是简单异或和。

如果有一个环，那么可以异或上这个环。

如果有多条路径，可以有一条路径异或上其他的环得到另一条路径。


### [Minimum spanning tree for each edge](https://www.luogu.com.cn/problem/CF609E)

树链剖分乱搞查询每一条边。
```cpp
int n,m;
vector<pii>g[N];
struct treelca{
    int siz[N],son[N],fa[N],dep[N],top[N];
    int a[N];
	inline void dfs1(int u,int from){
        siz[u]=1;son[u]=0;
        dep[u]=dep[from]+1;
        for(auto [v,w]:g[u]){
            if(v==from) continue;
            a[v]=w;
			fa[v]=u;dfs1(v,u);
            if(siz[v]>siz[son[u]])son[u]=v;
            siz[u]+=siz[v];
        }
    }
	int idx[N],cnt,dfn[N];
    inline void dfs2(int u,int tp){
        top[u]=tp;idx[u]=++cnt;
		dfn[cnt]=u;
        if(son[u]!=0) dfs2(son[u],tp);
        for(auto [v,w]:g[u]){
            if(v==fa[u]||v==son[u]) continue;
            dfs2(v,v);
        }
    }
    inline int lca(int x,int y){
        while(top[x]!=top[y]){
            if(dep[top[x]]>dep[top[y]])x=fa[top[x]];
            else y=fa[top[y]];
        }
        return dep[x]<dep[y]?x:y;
    }
	int tr[N<<2];
	inline void push_up(int k){
		tr[k]=max(tr[lc],tr[rc]);
	}
	inline void build(int k=1,int l=1,int r=n){
		if(l==r){
			tr[k]=a[dfn[l]];
			return;
		}
		int mid=(l+r)>>1;
		build(lc,l,mid);
		build(rc,mid+1,r);
		push_up(k);
	}
	inline int ask(int x,int y,int k=1,int l=1,int r=n) {
    	if (x <= l && r <= y) return tr[k];
    	int mid=(l+r)>>1,res=0;
    	if (x<=mid)res=max(res,ask(x,y,lc,l,mid));
    	if (y>mid)res=max(res,ask(x,y,rc,mid+1,r));
    	return res;
	}
	inline int qmax(int x, int y) {
    	int res=0; 
    	while(top[x]^top[y]){
        	if(dep[top[x]]<dep[top[y]]) swap(x, y);
        	res=max(res,ask(idx[top[x]],idx[x]));
        	x=fa[top[x]];
    	}
    	if(dep[x]>dep[y])swap(x, y);
    	return max(res,ask(idx[x]+1,idx[y]));
	}
}T;
struct edge{
	int u,v,w,id;
}e[N<<1];
inline bool cmp(edge x,edge y){
	return x.w<y.w;
}
inline bool cmp2(edge x,edge y){
	return x.id<y.id;
}
int fa[N];
inline int find(int x){
	if(x==fa[x])return x;
	return fa[x]=find(fa[x]);
}
bool vis[N];
signed main(){
	n=read();m=read();
	int u,v,w;
	up(i,1,m){
		e[i].u=read();
		e[i].v=read();
		e[i].w=read();
		e[i].id=i;
	}
	sort(e+1,e+1+m,cmp);
	int ans=0;
	up(i,1,n)fa[i]=i;
	up(i,1,m){
		int fu=find(e[i].u),fv=find(e[i].v);
		if(fu==fv)continue;
		vis[e[i].id]=1;
		fa[fu]=fv;
		g[e[i].u].push_back({e[i].v,e[i].w});
		g[e[i].v].push_back({e[i].u,e[i].w});
		ans+=e[i].w;
	}
	T.dfs1(1,0);
	T.dfs2(1,0);
	T.build();
	sort(e+1,e+1+m,cmp2);
	up(i,1,m) {
        if(vis[i])write(ans,1);
		else write(ans+e[i].w-T.qmax(e[i].u,e[i].v),1);
    }
	return 0;
}
```

### [Almost Acyclic Graph](https://www.luogu.com.cn/problem/CF915D)

有意思的暴力乱搞题。

因为 $n\le500$,所以可以考虑暴力减少每个点的度数。

```cpp
int n,m;
vector<int>g[N];
bool vis[N];
int deg[N],t[N];
inline bool top(){
	int cnt=0;
	queue<int> q;
	up(i,1,n)if(!deg[i])q.push(i);
	while(q.size()){
		int u=q.front();q.pop();
		cnt++;
		for(auto v:g[u])if(--deg[v]==0)	q.push(v);
	}
	return cnt==n;
}
signed main(){
	n=read();m=read();
	int u,v;
	up(i,1,m){
		u=read();v=read();
		g[u].push_back(v);
		deg[v]++;
	}
	up(i,1,n) t[i]=deg[i];
	up(i,1,n){
		if(deg[i]){
			deg[i]--;
			if(top())	return puts("YES"),0;
		}
		up(i,1,n)deg[i]=t[i];
	}
	puts("NO");
	return 0;
}
```

### [Connected Components?](https://www.luogu.com.cn/problem/CF920E)

暴力用 set 乱搞。

把每个点都加到 set 里面，然后对每个点进行 bfs。

搜到的点可以删除。

```cpp
int n,m;
set<int>g[N];
set<int>p;
queue<int>q;
inline int bfs(int u){
	q.push(u);
	int siz=0;
	while(q.size()){
		int u=q.front();q.pop();
		siz++;
		for(auto it=p.begin();it!=p.end();){
			if(g[u].count(*it)==0){
				q.push(*it);
				auto del=it;
				it++;
				p.erase(del);
			}
			else it++;
		}
	}
	return siz;
}
int ans[N];
signed main(){
	n=read();m=read();
	up(i,1,n)p.insert(i);
	int u,v;
	up(i,1,m){
		u=read();v=read();
		g[u].insert(v);
		g[v].insert(u);
	}
	int top=0;
    while(!p.empty()){
        u=*p.begin();
		p.erase(p.begin());
        ans[++top]=bfs(u);
    }
    sort(ans+1,ans+top+1);
    printf("%lld\n",top);
    up(i,1,top)printf("%lld ",ans[i]);
	return 0;
}
```

### [Trash Problem](https://www.luogu.com.cn/problem/CF1418D)

有关 set 的妙题。

答案可以表示为 $\max(p_n-p_1-(p_{k+1}-p_k))$
```cpp

set<int>s;
multiset<int>gap;
int n,q;
inline void insert(int x){
    s.insert(x);
    auto it=s.find(x);
    int pre=-1,nxt=-1;
    if(it!=s.begin()){
        gap.insert(x-*prev(it));
        pre=(*prev(it));
    }
    it++;
    if(it!=s.end()){
        gap.insert(*it-x);
        nxt=(*it);
    }
    if(pre!=-1&&nxt!=-1)gap.erase(gap.find(nxt-pre));
}
inline void del(int x){
    auto it=s.find(x);
    int pre=-1,nxt=-1;
    if(it!=s.begin()){
        gap.erase(gap.find(x-*prev(it)));
        pre=*prev(it);
    }
    it++;
    if(it!=s.end()){
        gap.erase(gap.find(*it-x));
        nxt=*it;
    }
    s.erase(x);
    if(pre!=-1&&nxt!=-1)gap.insert(nxt-pre);
}
int ans;
signed main(){
	n=read();q=read();
    int x;
    up(i,1,n){
        x=read();
        insert(x);
    }
    if(gap.size())ans=*prev(s.end())-*s.begin()-*prev(gap.end());
	cout<<ans<<"\n";
    int op;
    while(q--){
        op=read();x=read();
        if(op==0)del(x);
        else insert(x);
        ans=0;
        if(gap.size())ans=*prev(s.end())-*s.begin()-*prev(gap.end());
	    cout<<ans<<"\n";
    }
    return 0;
}
```
