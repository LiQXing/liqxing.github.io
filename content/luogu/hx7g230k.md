### [Peaceful Rooks](https://www.luogu.com.cn/problem/CF1411C)

因为棋子是车，所以移动的先后顺序没有关系。

就从前往后移动就行了。

```cpp
int n,m;
int fa[N];
inline int find(int x){
    if(x==fa[x])return x;
    return fa[x]=find(fa[x]);
}
signed main(){
    int T=read();
    while(T--){
        n=read();m=read();
        up(i,1,n)fa[i]=i;
        int ans=m,u,v;
        up(i,1,m){
            u=read();v=read();
            if(u==v)ans--;
            else if(find(u)==find(v)){
                ans++;
            }
            else fa[find(u)]=find(v);
        }
        write(ans,1);
    }
    return 0;
}
```

### [Cow and Snacks](https://www.luogu.com.cn/problem/CF1209D)

  
考虑一个大小$>1$的联通块。肯定有一个动物得吃到 $2$ 种点心，而总有一种排队顺序使得这个块中没有任何一头动物再吃到 $2$ 种点心，那么一个大小为  $c$ 的联通块可以满足 $c−1$ 个动物。

```cpp
int n,m;
int fa[N];
inline int find(int x){
    if(x==fa[x])return x;
    return fa[x]=find(fa[x]);
}
signed main(){
    n=read();m=read();
    up(i,1,n+m)fa[i]=i;
    int u,v;
    up(i,1,m){
        u=read();v=read();
        fa[find(i)]=find(u+m);
        fa[find(i)]=find(v+m);
    }
    int num=0;
    up(i,1,n+m){
        if(fa[i]==i)num++;
    }
    cout<<m-n+num;
    return 0;
}
```

###  [Masha-forgetful](https://www.luogu.com.cn/problem/CF1624E)
考虑将所有的可以记住的字符串分成若干个长度为 $2$ 或者 $3$ 的字串，并用一个哈希记录下来，然后 DP。
```cpp
struct str{
	int l,r,id;
}t[3005];
int T,n,a[1005][1005],m,fr[1005],f[1005],ans[1005],b[1005],rec[1005],cnt;
int hash(int num1,int num2,int num3){
	if(num1==0)num1=10;if(num2==0)num2=10;if(num3==0)num3=10;
	if(num1==-1)num1=0;
	return num1*121+num2*11+num3;
}
void init(){
	cnt=0;
	for(int i=1;i<=3000;i++)t[i].id=t[i].l=t[i].r=0;
	for(int i=1;i<=1000;i++)fr[i]=f[i]=ans[i]=b[i]=rec[i]=0;
}
int main(){
	scanf("%d",&T);
	while(T--){
		init();
		scanf("%d%d",&n,&m);
		for(int i=1;i<=n;i++){			
			for(int j=1;j<=m;j++)
			scanf("%1d",&a[i][j]);
			
			for(int j=2;j<=m;j++){
				int num=hash(-1,a[i][j-1],a[i][j]);
				t[num].l=j-1;t[num].r=j;t[num].id=i;
			}
			for(int j=3;j<=m;j++){
				int num=hash(a[i][j-2],a[i][j-1],a[i][j]);
				t[num].l=j-2;t[num].r=j;t[num].id=i;
			}
		}
		for(int i=1;i<=m;i++)scanf("%1d",&b[i]);
		f[0]=1;
		for(int i=2;i<=m;i++){
			if(f[i-2]){
				int num=hash(-1,b[i-1],b[i]);
				if(t[num].id!=0){
					f[i]=f[i-2]+1;fr[i]=i-2;ans[i]=num;
				}
			}
			if(i==2)continue;
			if(f[i-3]){
				int num=hash(b[i-2],b[i-1],b[i]);
				if(t[num].id!=0){
					f[i]=f[i-3]+1;fr[i]=i-3;ans[i]=num;
				}
			}
		}
		if(f[m]==0){
			printf("-1\n");
			continue;
		}
		printf("%d\n",f[m]-1);
		int now=m;
		while(now){
			rec[++cnt]=now;
			now=fr[now];
		}
		for(int i=cnt;i>=1;i--)
		printf("%d %d %d\n",t[ans[rec[i]]].l,t[ans[rec[i]]].r,t[ans[rec[i]]].id);
	}
	return 0;
}
```

### [Dominoes](https://www.luogu.com.cn/problem/CF267B)

单词接龙。

```cpp
int n,m;
int a[50][50];
int in[50];
int x[550],ansx[550],ansy[550],y[550];
int cnt;
inline void dfs(int u){
    up(i,0,6){
        if(a[u][i]>0){
            a[u][i]--;a[i][u]--;
            dfs(i);
            ansx[++cnt]=u;
            ansy[cnt]=i;
        }
    }
}
signed main(){
	cin>>n;
    up(i,1,n){
        cin>>x[i]>>y[i];
        a[x[i]][y[i]]++;a[y[i]][x[i]]++;
        in[x[i]]++;in[y[i]]++;
    }
    int j=0;
    up(i,0,6)if(in[i]>0)j=i;
    up(i,0,6)if((in[i]&1)==1)j=i,cnt++;
    if (cnt==1||cnt>2){
        printf("No solution\n");
        return 0;
    }
    cnt=0;
    dfs(j);
    if(cnt<n){
        printf("No solution\n");
        return 0;
    }
    dn(i,cnt,1){
        up(j,1,n){
            if (ansx[i]==x[j] && ansy[i]==y[j]){
                printf("%d +\n",j);
                x[j]=7;
                break;
            }
            else if (ansx[i]==y[j] && ansy[i]==x[j]){
                printf("%d -\n",j);
                x[j]=7;
                break;
            }
        }
    }
    return 0;
}
```

### [Cycle in Graph](https://www.luogu.com.cn/problem/CF263D)

就是暴力找环吗。

```cpp

int n,m,k;
vector<int>g[N];
int vis[N],pos;
inline void dfs(int u,int tim){
	vis[u]=tim;
	for(auto v:g[u]){
		if(!vis[v]){
			dfs(v,tim+1);
			break;
		}
		if(vis[u]>=vis[v]+k){
			cout<<vis[u]-vis[v]+1<<endl;
			cout<<u<<" ";
			pos=v;
			return;
		}
	}
	if(pos!=-1)cout<<u<<" ";
	if(pos==u)pos=-1;
}
signed main(){
	cin>>n>>m>>k;
    up(i,1,m){
		int u,v;
		cin>>u>>v;
		g[u].push_back(v);
		g[v].push_back(u);
	}
	dfs(1,1);
    return 0;
}
```

### [Graph Cutting](https://www.luogu.com.cn/problem/CF405E)

在树上的做法很显然。

把图的 dfs 树建出来，做法是一样的。
```cpp
int n,m;
vector<int>g[N];
vector<int>f[N];
bool vis[N];
inline void dfs(int u,int from){
    vis[u]=1;
    for(auto v:g[u]){
        if(vis[v]){
        	if(v!=from)f[v].push_back(u);
            continue;
        }
        dfs(v,u);
        f[u].push_back(v);
        if(f[v].size()){
            printf("%d %d %d\n",f[v][0],v,u);
            f[v].pop_back();
            f[u].pop_back();
        }
    }
    int siz=f[u].size()-1;
    while((siz+1)>1){
        printf("%d %d %d\n",f[u][siz],u,f[u][siz-1]);
        siz-=2;
        f[u].pop_back();
        f[u].pop_back();
    }
}
signed main(){
	cin>>n>>m;
    if(m&1){
        puts("No solution");
        return 0;
    }
    int u,v;
    up(i,1,m){
        cin>>u>>v;
        g[u].push_back(v);
        g[v].push_back(u);
    }
    dfs(1,0);
    return 0;
}
```

### [Wormhouse](https://www.luogu.com.cn/problem/CF62D)

用类似数位 DP 的方法来处理。

```cpp
int n,m;
int g[550][550];
int a[N],ans[N];
inline bool dfs(int u,int w,int lim){
    ans[u]=w;
    if(u==m+1&&w==a[m+1]&&!lim)return 1;
    if(u==m+1){
        ans[u]=0;
        return 0;
    }
    up(i,1,n){
		if((!g[w][i])||(lim&&i<a[u+1])) continue;
		g[w][i]=g[i][w]=0;
		if(dfs(u+1,i,lim&&(i==a[u+1])))return 1;
		g[w][i]=g[i][w]=1;
	}
	ans[u]=0;
    return 0;
}
signed main(){
    cin>>n>>m;
    int u,v;
    up(i,1,m+1){
        cin>>a[i];
        if(i<=1)continue;
        g[a[i]][a[i-1]]++;
        g[a[i-1]][a[i]]++;
    }
    if(dfs(1,a[1],1))up(i,1,m+1)cout<<ans[i]<<" ";
    else puts("No solution");
    return 0;
}
```
