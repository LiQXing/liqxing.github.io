## [P3045 [USACO12FEB] Cow Coupons G](https://www.luogu.com.cn/problem/P3045)

虽然看起来是一个背包，但是看到 $m\le10^{14}$ 就应该排除。

考虑到用优惠券一定使价格变低。

所以前 $k$ 小的 $c$ 一定是最优的。

如果需要增加奶牛，有两种做法，一种是在剩下的奶牛里面调出 $p$ 最小的。
一种是在前面找出 $p$ 与 $c$ 差值最小的，然后从后面调出 $c$ 最小的。

所以只用维护3个小根堆就行了。

```cpp
int n,k,m;
struct node{
	int p,c;
}a[N];
inline bool cmp(node x,node y){
	return x.c<y.c;
}
bool vis[N];
priority_queue<pii,vector<pii>,greater<pii> >q1,q2,q3;
signed main(){
	n=read();k=read();m=read();
	up(i,1,n){
		a[i].p=read();
		a[i].c=read();
	}
	sort(a+1,a+1+n,cmp);
	int now=0;
	up(i,1,k){
		now+=a[i].c;
		if(now>m){
			cout<<i-1;
			return 0;
		}
		q1.push({a[i].p-a[i].c,i});
	}
	up(i,k+1,n){
		q2.push({a[i].c,i});
		q3.push({a[i].p,i});
	}
	int ans=k;
	up(i,k+1,n){
		while(vis[q2.top().second])q2.pop();
		while(vis[q3.top().second])q3.pop();
		int pos2=q2.top().second,pos3=q3.top().second;
		int tmp2=q2.top().first+q1.top().first,tmp3=q3.top().first;
		if(tmp2<tmp3){
			now+=tmp2;
			q1.pop();q2.pop();
			q1.push({a[pos2].p-a[pos2].c,pos2});
			vis[pos2]=1;
		}
		else{
			now+=tmp3;
            q3.pop();
            vis[pos3]=1;
		}
		ans++;
		if(now>m){
            cout<<ans-1<<endl;
            return 0;
        }
	}
	cout<<n;
	return 0;
}
```

## [P1484 种树](https://www.luogu.com.cn/problem/P1484)

一个$O(nk)$ 复杂度的暴力DP是很容易想的。

然而正解却是贪心。

当 $k=1$ 时，显然是取最大的 $a[i]$。
当 $k=2$ 时，要么取$a[i]$ 和 $\max_{j<(i-1)||j>(i+1)}(a[j])$，要么取 $a[i-1],a[i+1]$。

那么我们可以考率，当从优先队列取出了一个点后，我们再往里面加一个点，权值为 $a[i+1]+a[i-1]-a[i]$。

```cpp
int n,k;

bool vis[N];
int a[N],l[N],r[N],ans;
priority_queue<pii>q;
signed main(){
	n=read();k=read();
	up(i,1,n){
		a[i]=read();
		q.push({a[i],i});
		l[i]=i-1;r[i]=i+1;
	}
	l[n+1]=n;r[0]=1;
	up(i,1,k){
		while(vis[q.top().second])q.pop();
		t=q.top(); q.pop();
		if(t.first<0) break;
		ans+=t.first;
		int pos=t.second;
		a[pos]=a[l[pos]]+a[r[pos]]-a[pos];
		t.first=a[pos]; 
		vis[l[pos]]=vis[r[pos]]=1;
		l[pos]=l[l[pos]];
		r[pos]=r[r[pos]];
		r[l[pos]]=pos;
		l[r[pos]]=pos; 
		q.push(t);
	}
	cout<<ans;
	return 0;
}
```

```cpp
		int x=read();
		if(!q.empty()&&q.top()<x)ans+=x-q.top(),q.pop(),q.push(x);
		q.push(x);
	}
```

## [P2048 [NOI2010] 超级钢琴](https://www.luogu.com.cn/problem/P2048)

非常棒的一道题。

我们先用 RMQ求出每个满足条件的区间。

然后如果一个区间被选了，就加入分割的两个区间。

```cpp
struct ST{
    int dp[1][20],pos[N][20];
    inline void build(int l,int r,int arr[]){
       // up(i,l,r)dp[i-l+1][0]=arr[i];
        up(i,l,r)pos[i][0]=i;
        int res=0;
        for(int j=1;(1<<j)<=r-l+1;j++){
            for(int i=l;i+(1<<j)-1<=r;i++){
                int x=pos[i-l+1][j-1],y=pos[i-l+1+(1<<(j-1))][j-1];
                //dp[i-l+1][j]=max(dp[i-l+1][j-1],dp[i-l+1+(1<<(j-1))][j-1]);
                pos[i-l+1][j]=arr[x]>arr[y]?x:y;
            }
        }
    }
    inline int ask(int l,int r){
        int k=log2(r-l+1);
        return max(dp[l][k],dp[r-(1<<k)+1][k]);
    }
    inline int askpos(int l,int r,int arr[]){
        int k=log2(r-l+1);
        int x=pos[l][k],y=pos[r-(1<<k)+1][k];
        return arr[x]>arr[y]?x:y;
    }
}T;
int n,k,l,r;
int a[N],sum[N];
struct node{
    int o,l,r,t;
    inline bool operator <(const node&rhs)const{
        return sum[t]-sum[o-1] <sum[rhs.t]-sum[rhs.o-1] ;
    }
};
priority_queue<node>q;
signed main(){
    n=read();k=read();l=read();r=read();
    up(i,1,n){
        a[i]=read();
        sum[i]=sum[i-1]+a[i];
    }
    T.build(1,n,sum);
    up(i,1,n){
        if(i+l-1<=n){
            q.push({i,i+l-1,min(i+r-1,n),T.askpos(i+l-1,min(i+r-1,n),sum)});
            //cout<<T.askpos(i+l-1,min(i+r-1,n),sum)<<endl;
        }
    }
    int ans=0;
    while(k--){
        node u=q.top();q.pop();
        ans+=sum[u.t]-sum[u.o-1];
        if(u.t!=u.l)q.push({u.o,u.l,u.t-1,T.askpos(u.l,u.t-1,sum)});
        if(u.t!=u.r)q.push({u.o,u.t+1,u.r,T.askpos(u.t+1,u.r,sum)});
    }
    cout<<ans;
```

```cpp
            i=q2.top().second;
            q2.pop();
        }
        i<n&&(q1.push({a[i+n],i+n}),1);
        ans+=a[i];
        star[i%n]++;
        vis[i]=1;
        //cout<<ans<<endl;
    }
    cout<<ans<<endl;
    up(i,1,n) cout<<star[i];
    return 0;
}
```

## [Difficult Mountain](https://www.luogu.com.cn/problem/CF1601D)

非常离谱的一道贪心。

直接上结论：

人按$\max(a_i​,s_i​)$ 从小到大排序。如果 $\max(a_i​,s_i​)$ 相等，则按 $s_i$​​ 从小到大排序。

为什么？

当遍历到第 $i$ 个人的时候

 ### 1.
 $\max(a_i​,s_i​)$ 为 $s_i$​ 时，显然$s_i​≥d$，因为在 $i$ 之前的人只有两种：

-   $\max(s_j​,a_j​)<s_i$​ 时，显然 $d<s_i$​。
-   $\max(s_j​,a_j​)=s_i$​​​ 时，显然 $d≤s_i​$。

所以这种情况的话就能更新答案。

### 2.

$\max(a_i​,s_i​$) 为 $a_i$​ 时，用前文的证明方式可得：$d\le a_i$​。

此时如果 $s_i​≥d$ ，显然直接更新答案了，证明同上。

否则不更新答案。显然如果非要让这个人去，**至少**也要让之前的一个人不去，总答案不变，但 $d$​ 不会变得更小，所以最好的策略是不去。

## [Guard Duty (medium)](https://www.luogu.com.cn/problem/CF958E2)

排完序就是一道种树。

```cpp
priority_queue<pii,vector<pii>,greater<pii> >p;
int n,k,a[N],L[N],R[N],len,vis[N],ans;
signed main(){
	k=read();n=read();
	up(i,1,n)a[i]=read();
	sort(a+1,a+n+1);
	up(i,1,n-1){
		a[i]=a[i+1]-a[i]; 
		p.push({a[i],i});
		L[i]=i-1,R[i]=i+1;
	} 
	a[0]=a[n]=inf;
	len=n;
	while(k--){
		while(!p.empty()&&vis[p.top().second])p.pop();
		int t=p.top().second,v=p.top().first;
		ans+=v;
		p.pop();
		int l=L[t],r=R[t];
		vis[t]=1;
        vis[l]=1;
        vis[r]=1;
		a[++len]=a[r]+a[l]-a[t];
		L[len]=L[l],R[len]=R[r],R[L[l]]=len,L[R[r]]=len;
		p.push({a[len],len});
	}
	cout<<ans;
    return 0;
}
```

## [Jeff and Permutation](https://www.luogu.com.cn/problem/CF351E)

-   然后我们考虑对于一组$(i,j),i<j$ 会产生逆序对的情况：
    
    -   $Case1\quad a_i<a_j$
        
        那么要产生逆序对只能：(1)让$a_j$​取反，$a_i$不变，(2)让$a_i,a_j$都取反
        
    -   $Case2\quad a_i>a_j$
        
        那么要产生逆序对只能：(1)让$a_i$,$a_j$​都不取反，(2)$a_i$不取反，$a_j$取反
        
-   那么综上两种情况：如果$a_i$不选择取反，那么就是第二种情况（后面的数比他小），否则$a_i$取反那么就是前面比他小的数都会产生贡献。然后就对左右两边统计比$a_i$小的个数取$\min$。
