## [Clique Problem](https://www.luogu.com.cn/problem/CF527D)

很有意思的一道题。

令 $x_i>x_j$ ,  则有$x_i-w_i\ge x_j+w_j$

对每一个点，都需要 判断每一个点的区间 $[x_i-w_i,x_i+w_i]$

  
我们可以把每个点都看成是 $[l,r]$ 区间上的一条线段，如果满足刚才的那个式子，就说明两条线段不重叠（因为一个的最左端在另一个最右端的右侧，肯定不重叠），所以任意两条互不重叠的线段之间都会相连。

## [Portals](https://www.luogu.com.cn/problem/CF1271D)

设 $f[i][j]$ 表示攻下了前 $i$ 个城堡且手上还剩 $j$ 个士兵时，被控制的最大总价值。那么这个问题就类似一个背包，直接转移一下就可以得到答案了。只要注意边界——打第 $i$ 个城堡前手上必须有至少 $a[i]$ 个士兵。

```cpp
int n,m,k;
int a[N],b[N],c[N],pre[N];
int f[N];
vector<int>g[N];
signed main() {
    n=read();m=read();k=read();
    up(i,1,n){
        a[i]=read();b[i]=read();c[i]=read();
        pre[i]=i;
    }
    int u,v;
    up(i,1,m){
        u=read();v=read();
        pre[v]=max(pre[v],u);
    }
    up(i,1,n){
        g[pre[i]].push_back(i);
    }
    fill(f,f+N,-inf);
    f[k]=0;
    int ans = -inf;
    up(i,1,n){
        for(int j = (k += b[i]); (j - b[i]) >= a[i]; --j) {
            f[j] = f[j - b[i]];
        }
        fill(f, f + a[i],-inf);
        for (auto u :g[i]) {
            for (int j = 0, dj = j + 1; j <k; j = dj++) {
                f[j]=max(f[j], f[dj] + c[u]);
            }
        }
    }
    for (int i = 0; i <=k; ++i) {
        ans = max(ans, f[i]);
    }
    if(ans<0)cout<<-1;
    else cout<<ans;
    return 0;
}
```

## [You Are Given a WASD-string... ](https://www.luogu.com.cn/problem/CF1202C)

看到这道题 第一反应是行列减一。
原本面积是 $a\times b$,答案应该是 $(a-1)\times b,a\times(b-1)$.

然而这个结论是错误的。

但是考虑这样一个操作序列 `ADAD` 原本是 $1×2$，但任意添加操作之后并不能让权值变为$ 0×2$，甚至不能变为 $1×1$。

那么变化的位置只能是 **一个最值最后一次出现与另一个最值第一次出现** 之间。

```cpp
int n;
int sum[N],sum2[N];
inline bool check(int l,int r){
  for(int i=l+1;i<=r;++i)if(sum[i]!=sum[l])return true;
  return false;
}
inline bool check2(int l,int r){
  for(int i=l+1;i<=r;++i)if(sum2[i]!=sum2[l])return true;
  return false;
}
string s;
signed main() { 
	int T=read();
	while(T--){
		cin>>s;
		n=s.size();s=" "+s;
		int maxl=0,maxl2=0,minl=0,minl2=0;
		up(i,1,n){
      		sum[i]=sum[i-1];
      		if(s[i]=='A')--sum[i];
      		if(s[i]=='D')++sum[i];
      		maxl=max(maxl,sum[i]);minl=min(minl,sum[i]);
      		sum2[i]=sum2[i-1];
      		if(s[i]=='S')--sum2[i];
      		if(s[i]=='W')++sum2[i];
      		maxl2=max(maxl2,sum2[i]);minl2=min(minl2,sum2[i]);
		}
    	int l=1e14,r=-1e14,l2=1e14,r2=-1e14;
    	for(int i=0;i<=n;++i){
      		if(sum[i]==minl)l=min(l,i),r=max(r,i);
      		if(sum[i]==maxl)l2=min(l2,i),r2=max(r2,i);
    	}
    	int tot=(maxl2-minl2+1)*(maxl-minl+1),ans=tot;
    	if((l>r2||r<l2)&&(check(r2,l-1)||check(r,l2-1)))ans=min(ans,tot-(maxl2-minl2+1));
    	l=l2=1e14;r=r2=-1e14;
    	up(i,0,n) {
      		if(sum2[i]==minl2)l=min(l,i),r=max(r,i);
      		if(sum2[i]==maxl2)l2=min(l2,i),r2=max(r2,i);
    	}
    	if((l>r2||l2>r)&&(check2(r2,l-1)||check2(r,l2-1)))ans=min(ans,tot-(maxl-minl+1));
    	cout<<ans<<endl;
	}
    return 0;
}
```
## [Unusual Sequences ](https://www.luogu.com.cn/problem/CF900D)

很有意思的一道数学题。

等价为求一个和为$\dfrac{y}{x}$​ 的序列，使得其 $\gcd$ 为 $1$

和为 $s$ 的数列有 $2^{s-1}$,和为 $s,\gcd =1$的数列个数为 $f(s)$。

所以有$$f(k)=2^{k-1}-\sum_{d|k,d>1}f(\dfrac{k}{d})$$

```cpp
inline int ksm(int a,int b){
	int res=1;
	while(b){
		if(b&1)res=res*a%mod;
		a=a*a%mod;
		b>>=1;
	}
	return res;
}
unordered_map<int,int>mp;
inline int dfs(int n) {
	if(n==1) return 1;
	if(mp.count(n)) return mp[n];
	int ans=0;
	for(int i=2;i*i<=n;++i) {
		if(n%i==0) {
			ans=(ans+dfs(i))%mod;
			if(i*i!=n) ans=(ans+dfs(n/i))%mod;
		}
	}
	ans=(ans+dfs(1))%mod;
	ans=(ksm(2,n-1)-ans+mod)%mod;
	return mp[n]=ans;
}
signed main() { 
	x=read();y=read();
	if(y%x){
		printf("0\n");
		return 0;
	}
	cout<<dfs(y/x);
    return 0;
}
```

## [Photo Processing](https://www.luogu.com.cn/problem/CF883I#submit)

不难想到排序，因为想要最小化每组中的最小值，那每组中的数必然是挨着的，比如 55 个数被分为 $[1,3][4,5][9]$ ，假设我们交换了3,5 必然造成两个组的权值都增加，这明显是不优的，因此我们只需要考虑连续区间即可。
```cpp
int n,m;
int a[N];
int f[N];
inline bool check(int x){
	f[0]=1;
	fill(f+1,f+1+n,0);
	int l=1,r=0;
	for(int i=1;i<=n;i++){
		while(a[i]-a[l]>x)l++;
		r=i-m+1;
		while(!f[l-1]&&l<=r)l++;
		if(l<=r)f[i]=1;  
	}

}
int ans;
```

## [Lunar New Year and Red Envelopes](https://www.luogu.com.cn/problem/CF1106E)

题目的关键在$s[i]\le t[i]\le d[i]$。

所以如果能行动，则该区间的红包一定能收集。

```cpp
int n,m,k;
struct node{
    int s,t,d,w;
    bool operator<(const node &x)const {
    	if(w == x.w) return d < x.d;
        return w < x.w;
    }
}a[100500];
priority_queue<node>q;
inline bool cmp(node x,node y){
    return x.s<y.s;
}
int g[N],f[100500][205],h[N];
signed main(){
    n=read();m=read();k=read();
    up(i,1,k){
        a[i].s=read();
        a[i].t=read();
        a[i].d=read();
        a[i].w=read();
    }
    sort(a+1,a+1+k,cmp);
    int cur=1;
    g[0]=1;
    up(i,1,n){
        while(cur<=k&&a[cur].s<=i)q.push(a[cur++]);
        if(q.empty())g[i]=i+1;
        else{
            node u = q.top();
            if(u.t<i){
                i--;
                q.pop();
                continue;
            }
            g[i]=u.d+1;
            h[i]=u.w;
        }
    }
    memset(f,0x3f,sizeof f);
    f[0][0]=0;
    up(i,0,n){
        up(j,0,m){
            f[i+1][j+1]=min(f[i+1][j+1],f[i][j]);
            f[g[i]][j]=min(f[g[i]][j],f[i][j]+h[i]);
        }
    }
    int ans=inf;
    up(i,0,m)ans=min(ans, f[n+1][i]);
    cout<<ans;
    return 0;
}
```
